# Compilacion para windows usando pyInstaller (onefile) o (ondir)

pyinstaller --onefile --console --name "NdxYtConver" --icon "icon.ico" --paths=. --add-data "core:core" cli/cliWinVer.py

# Instalarlo como libreria de dependecias (uso para COMANDOS)
pip install e .

# En caso que te de problemas luego de instalar paquetes
